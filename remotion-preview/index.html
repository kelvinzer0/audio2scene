Composition Code (excerpt)

The Remotion composition that turns audio2scene events into video. Full source: github.com/kelvinzer0/audio2scene

// src/Composition.tsx — audio2scene → Remotion import { AbsoluteFill, Audio, staticFile, useCurrentFrame, useVideoConfig } from "remotion"; export const MyComposition = () => { const frame = useCurrentFrame(); const { fps } = useVideoConfig(); const currentTime = frame / fps; // Find active segment + effect at current time const segment = findSegment(song.segments, currentTime); const flashEvent = song.events.find( (e) => e.effect === "Flash" && currentTime >= e.time && currentTime < e.time + 0.1 ); const glitchEvent = song.events.find( (e) => e.effect === "Glitch" && currentTime >= e.time && currentTime < e.time + 1.0 ); return ( <AbsoluteFill style={{ backgroundColor: "#0a0a0f" }}> <Audio src={staticFile("audio.mp3")} /> <SceneBackground segment={segment} intensity={segment.intensity} /> <FlashOverlay active={!!flashEvent} intensity={flashEvent?.intensity} /> <GlitchEffect active={!!glitchEvent} intensity={glitchEvent?.intensity} /> <SegmentLabel segment={segment} /> <Timeline events={song.events} currentTime={currentTime} /> </AbsoluteFill> ); };

Render Commands

How to reproduce this render locally.

# 1. Install audio2scene + run analysis pip install -e . audio2scene song.mp3 --video-editor -o events.json # 2. Scaffold Remotion project npx create-video@latest --yes --blank --no-tailwind my-video cd my-video npx remotion add @remotion/media-utils # 3. Place audio + events in public/ cp song.mp3 public/audio.mp3 cp events.json public/song.json # 4. Open Remotion Studio (interactive preview) npx remotion studio # 5. Render to MP4 npx remotion render Audio2ScenePreview out/preview.mp4 --frames=0-900
